	<head>
		<meta charset="utf-8" />
		<title>Mapbox GL Legend example</title>
		<meta name="viewport" content="initial-scale=1,maximum-scale=1,user-scalable=no" />

		<script type="importmap">
			{
				"imports": {
					"mapbox-gl": "https://esm.sh/mapbox-gl@^3.0.0"
				}
			}
		</script>
		<link href="https://esm.sh/mapbox-gl@^3.0.0/dist/mapbox-gl.css" rel="stylesheet" />
		<link
			href="https://cdn.jsdelivr.net/npm/@watergis/mapbox-gl-legend@{version}/dist/mapbox-gl-legend.css"
			rel="stylesheet"
		/>

		<style>
			body {
				margin: 0;
				padding: 0;
			}
			#map {
				position: absolute;
				top: 0;
				bottom: 0;
				width: 100%;
			}
		</style>
	</head>

	<body>
		<div id="map"></div>

		<script type="module">
			import mapboxgl from 'mapbox-gl';
			import { MapboxLegendControl } from 'https://cdn.jsdelivr.net/npm/@watergis/mapbox-gl-legend@{version}/dist/mapbox-gl-legend.es.js';

			mapboxgl.accessToken = '{accesstoken}';

			const map = new mapboxgl.Map({
				container: 'map',
				style: '{style}',
				center: [{center}],
				zoom: {zoom}
			});

			map.on('load', () => {
				const targets = {targets};
				const options = {options};
				map.addControl(new MapboxLegendControl(targets, options), 'top-right');
			});
		</script>
	</body>
