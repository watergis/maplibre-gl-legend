				"imports": {

		<script type="module">
			import { Map } from 'maplibre-gl';
			import { MaplibreLegendControl } from 'https://cdn.jsdelivr.net/npm/@watergis/maplibre-gl-legend@{version}/dist/maplibre-gl-legend.es.js';

			const map = new Map({
				container: 'map',
				style: '{style}',
				center: [{center}],
				zoom: {zoom}
			});

			map.on('load', () => {
				const targets = {targets};
				const options = {options};
				map.addControl(new MaplibreLegendControl(targets, options), 'top-right');
			});
		</script>
